            Pause <i class="glyphicon glyphicon-pause"></i>
        </button>
        <button onclick="stopAll()" class="btn btn-danger">

    <script>
        // Seleccionar todos los elementos de audio
        const audioElements = document.querySelectorAll("audio");

        // Función para pausar todos los audios
        function pauseAll() {
            audioElements.forEach(audio => audio.pause());
        }

        // Función para detener y reiniciar todos los audios
        function stopAll() {
            audioElements.forEach(audio => {
                audio.pause();
                audio.currentTime = 0;
            });
        }

        // Función para rebobinar todos los audios
        function rewindAll() {
            audioElements.forEach(audio => {
                audio.currentTime = 0;
                audio.play();
            });
        }
    </script>
